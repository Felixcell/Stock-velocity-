Fetch Data df = yf.download(ticker, period=f"{days}d", interval=interval) if df.empty: return None, None, None df = df[['Open', 'High', 'Low', 'Close', 'Volume']].copy() # Flatten multi-index columns if generated by yfinance if isinstance(df.columns, pd.MultiIndex): df.columns = df.columns.get_level_values(0) # 2. Feature Engineering df['Return'] = df['Close'].pct_change() df['SMA_10'] = df['Close'].rolling(window=10).mean() df['SMA_30'] = df['Close'].rolling(window=30).mean() df['SMA_Ratio'] = df['SMA_10'] / df['SMA_30'] # RSI calculation delta = df['Close'].diff() gain = (delta.where(delta > 0, 0)).rolling(window=14).mean() loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean() rs = gain / (loss + 1e-9) df['RSI'] = 100 - (100 / (1 + rs)) # 3. Define Labels (1 if next close is higher, else 0) df['Target'] = np.where(df['Close'].shift(-1) > df['Close'], 1, 0) df.dropna(inplace=True) # Prepare datasets feature_cols = ['Return', 'SMA_Ratio', 'RSI'] X = df[feature_cols] y = df['Target'] # Standardize data scaler = StandardScaler() X_scaled = scaler.fit_transform(X) # Train Random Forest model = RandomForestClassifier(n_estimators=100, random_state=42) model.fit(X_scaled, y) return model, scaler, df #
