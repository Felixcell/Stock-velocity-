import yfinance as yf import pandas as pd def fetch_historical_data(ticker: str, period: str = "60d", interval: str = "5m") -> pd.DataFrame: """ Fetches historical stock data from Yahoo Finance. - period: "60d" (60 days of history) - interval: "5m" (5-minute bars for quick/short-term trading) """ print(f"Fetching data for {ticker}...") stock = yf.Ticker(ticker) df = stock.history(period=period, interval=interval) # Clean up column names to keep things standard df = df[['Open', 'High', 'Low', 'Close', 'Volume']] return df if __name__ == "__main__": # Test the downloader with Tesla (TSLA) test_data = fetch_historical_data("TSLA", period="5d", interval="5m") print(test_data.head()) print(f"Downloaded {len(test_data)} rows of 5-minute data.")
